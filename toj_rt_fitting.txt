# ============================
# Temporal RT Model (averaged data)
# ============================

library(rstanarm)
library(tidyverse)
library(parallel)

setwd("E:/TemporalSwitch/DataAnalysis/RT")
data <- read_csv("toj_rt_trials.csv")

# 🧼 Preprocess dataset: factor conversion, scaling, and filtering
data <- data %>%
  mutate(
    subj_idx = factor(subj_idx),
    hist = factor(hist),
    rt = as.numeric(rt),
    soa = as.numeric(soa/28 - mean(soa/28)),
    rsi = as.numeric(rsi*10 - mean(rsi*10)),
    block = as.numeric(block - mean(block))
  ) %>%
  drop_na(rt, soa, hist, block, rsi, subj_idx)

# 🔧 Set reference level for contrast coding
data$hist <- relevel(data$hist, ref = "R")

# 📊 Aggregate: compute average RT per subject × condition
data_avg <- data %>%
  group_by(subj_idx, hist, soa, rsi, block) %>%
  summarise(
    rt = mean(rt, na.rm = TRUE)*1000,
    .groups = "drop"
  )

# 💾 Save aggregated dataset
write_csv(data_avg, "toj_rt_data_avg.csv")
message("✅ Aggregated data saved as 'toj_rt_data_avg.csv'")

# 📊 Define weakly informative priors
prior_intercept <- normal(location = 0, scale = 1, autoscale = TRUE)
prior_slopes <- normal(location = 0, scale = 1, autoscale = TRUE)

# 🎲 Set seed
current_seed <- 12345

# 📈 Fit GLMM on averaged data
fit_rt_model_avg <- stan_glmer(
  rt ~ hist + soa + I(soa^2) + rsi + block +
    hist:soa + hist:I(soa^2) + hist:rsi + hist:block +
    (1 + hist | subj_idx),  # Random intercepts and slopes for 'hist' by subject
  data = data_avg,
  family = gaussian(link = "identity"),
  prior_intercept = prior_intercept,
  prior = prior_slopes,
  chains = 4,
  iter = 4000,
  warmup = 2000,
  cores = parallel::detectCores(),
  seed = current_seed,
  adapt_delta = 0.95,
  refresh = 100
)

# 💾 Save fitted model
saveRDS(fit_rt_model_avg, "toj_rt_model_fit_avg_stan_glmer.rds")
