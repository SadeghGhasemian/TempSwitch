# ===================== #
# 🔧 0. Setup
# ===================== #
library(brms)
library(tidyverse)
library(cmdstanr)
library(bayestestR)
library(posterior)

options(brms.backend = "cmdstanr")
set.seed(123)
 setwd("E:/TemporalSwitch/DataAnalysis/brms") # Use here::here() for reproducible paths

# ===================== #
# 📁 1. Load & Aggregate
# ===================== #
df_raw <- read_csv("toj_choice_trials.csv") %>%
  mutate(
    response = as.numeric(response == 1),
    soa_sec = soa/1000,
    hist = factor(hist),
    subj = factor(subj)
  )

df_binom <- df_raw %>%
  group_by(soa_sec, hist, subj) %>%
  summarise(
    successes = sum(response),
    trials = n(),
    .groups = "drop"
  )

write_csv(df_binom, "toj_trials_aggregated_binomial.csv")

# ===================== #
# 🧠 2. Nonlinear Model (Optimized)
# ===================== #
# Using the standard probit link simplifies the model and improves stability
nonlinear_formula_optimized <- bf(
 successes | trials(trials) ~ (soa_sec - PSS) / sigma,
  PSS  ~ 1 + hist + (1 + hist | subj),
  sigma ~ 1 + hist + (1 + hist | subj),
  nl = TRUE
)

# ===================== #
# 🎯 3. Priors (Refined)
# ===================== #
priors_refined <- c(
  prior(normal(0, 0.1), class = "b", nlpar = "PSS"),
  prior(exponential(1), class = "sd", nlpar = "PSS"),
  prior(normal(0.1, 0.05), class = "b", nlpar = "sigma", lb = 0.02),
  prior(exponential(1), class = "sd", nlpar = "sigma")
)

# ===================== #
# 🚀 4. Fit Model (Optimized)
# ===================== #
fit_brms_optimized <- brm(
  formula = nonlinear_formula_optimized,
  data = df_binom,
  prior = priors_refined,
  family = binomial(link = "probit"), # Use the standard probit link
  backend = "cmdstanr",
  chains = 4,
  cores = parallel::detectCores(),
  threads = threading(2),
  iter = 6000,
  warmup = 3000,
  init = 0,
  control = list(adapt_delta = 0.995, max_treedepth = 15),
  file = "brms_psychometric_fit_binomial_toj.rds"
)

setwd("E:/TemporalSwitch/DataAnalysis/brms")

# Load raw data and store original SOA mean before centering
df_raw <- read_csv("toj_choice_trials.csv") %>%
  mutate(
    response = as.numeric(response == 1),
    soa_sec = soa / 1000  # convert ms to sec
  )

soa_mean_sec <- mean(df_raw$soa_sec)      # mean in seconds
soa_mean_ms  <- soa_mean_sec * 1000       # mean in ms

# Create centered version for any later use
df <- df_raw %>%
  mutate(
    soa = soa_sec - soa_mean_sec,  # mean-centered
    hist = factor(hist),
    subj = factor(subj)
  )

# =========================== #
# 🔍 Extract Posterior Draws
# =========================== #
draws <- as_draws_df(fit_brms)

# =========================== #
# 🧮 Compute Condition-wise Estimates
# =========================== #
# PSS in centered seconds
draws$PSS_R <- draws$b_PSS_Intercept
draws$PSS_S <- draws$b_PSS_Intercept + draws$`b_PSS_histS`

# sigma (spread) in seconds
draws$sigma_R <- draws$b_sigma_Intercept
draws$sigma_S <- draws$b_sigma_Intercept + draws$`b_sigma_histS`

# =========================== #
# 🔁 Back-transform to milliseconds
# =========================== #
# PSS back to absolute ms scale
draws$PSS_R_ms <- (draws$PSS_R + soa_mean_sec) * 1000
draws$PSS_S_ms <- (draws$PSS_S + soa_mean_sec) * 1000

# JND = sigma * 0.674, also in ms
draws$JND_R_ms <- draws$sigma_R * 0.674 * 1000
draws$JND_S_ms <- draws$sigma_S * 0.674 * 1000

# =========================== #
# ➕ Differences between conditions
# =========================== #
draws$diff_PSS_ms <- draws$PSS_S_ms - draws$PSS_R_ms
draws$diff_JND_ms <- draws$JND_S_ms - draws$JND_R_ms

posterior_df <- draws %>%
  select(PSS_R_ms, PSS_S_ms, JND_R_ms, JND_S_ms,
         diff_PSS_ms, diff_JND_ms)

write_csv(posterior_df, "posterior_toj.csv")

ss <- describe_posterior(fit_brms)
print(ss, digits = 6)

rep <- abs(posterior_df$PSS_R_ms)
swit <- abs(posterior_df$PSS_S_ms)

diff <- swit-rep

diff_describe <- print(describe_posterior(diff), digits = 4)

rep <- posterior_df$JND_R_ms
swit <- posterior_df$JND_S_ms

diff <- swit-rep

diff_describe <- print(describe_posterior(diff), digits = 4)
